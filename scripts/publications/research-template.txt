<script>
class ScopusCitations extends HTMLElement {
  async connectedCallback() {
    const file = this.getAttribute("file");
    const label = this.getAttribute("label") || "SCOPUS-citations";
    const prefix = this.getAttribute("prefix") ?? (label + ": ");
    const missing = this.getAttribute("missing") || ""; // what to show if fetch fails
    const setText = text => {
      this.textContent = text;
      this.hidden = !text;
      const parent = this.parentElement;
      if (parent && parent.tagName === "P" && parent.childElementCount === 1) {
        parent.hidden = !text;
      }
    };

    if (!file) {
      setText(missing);
      return;
    }

    // Render a placeholder immediately
    setText(prefix + "…");

    try {
      // Resolve relative paths correctly (works with/without baseurl, subfolders, etc.)
      const url = new URL(file, window.location.href).toString();

      const r = await fetch(url, { cache: "no-store" });
      if (!r.ok) throw new Error(`HTTP ${r.status}`);

      const t = (await r.text()).trim();
      const m = t.match(/\d+/);          // raw file should be just digits, but be defensive
      if (!m) throw new Error("No number found");

      const count = Number(m[0]);
      setText(count === 0 ? "" : prefix + m[0]);
    } catch (e) {
      // Fail quietly
      setText(missing);
      // Uncomment for debugging:
      // console.warn("Scopus citation fetch failed:", file, e);
    }
  }
}

customElements.define("scopus-citations", ScopusCitations);
</script>
